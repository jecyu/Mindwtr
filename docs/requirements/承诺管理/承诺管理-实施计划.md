### 4.2 `packages/core/src/commitment-score.ts`（新建）

```ts
import type { Task } from './types';
import type { CommitmentBenchmark, CommitmentCard } from './commitment-types';

export type UrgencyReason = 'within-48h' | 'this-week' | 'no-deadline';

export interface ScorePart {
    name: string;
    value: number;
    note: string;
}

export interface CommitmentScore {
    goal: number;
    urgency: number;
    impact: number;
    delegable: number;
    value: number;
    benchmarkPoints: number;
    multiplier: number;
    complexity: number;
    p: number;
    urgencyReason: UrgencyReason;
    parts: ScorePart[];
}

const HOURS_48 = 48;
const HOURS_WEEK = 168;

/** 紧急分从 dueDate 派生 —— 48h 内 = 3 / 本周内 = 2 / 无硬截止 = 1 */
export function deriveUrgency(
    dueDate: string | undefined,
    now: Date,
): { value: number; reason: UrgencyReason } {
    if (!dueDate) return { value: 1, reason: 'no-deadline' };
    const due = new Date(dueDate).getTime();
    if (!Number.isFinite(due)) return { value: 1, reason: 'no-deadline' };
    const hours = (due - now.getTime()) / 3_600_000;
    if (hours <= HOURS_48) return { value: 3, reason: 'within-48h' };
    if (hours <= HOURS_WEEK) return { value: 2, reason: 'this-week' };
    return { value: 1, reason: 'no-deadline' };
}

/**
 * P = 价值分 ÷ 复杂度点。
 * 复杂度点 <= 0 时返回 null —— 该任务不参与 P 值排序。
 */
export function computeCommitmentScore(
    task: Pick<Task, 'dueDate'>,
    card: CommitmentCard,
    benchmark: CommitmentBenchmark | undefined,
    now: Date,
): CommitmentScore | null {
    const points = benchmark?.points;
    if (typeof points !== 'number' || !Number.isFinite(points) || points <= 0) return null;

    const complexity = points * card.multiplier;
    if (!Number.isFinite(complexity) || complexity <= 0) return null;

    const goal = Math.max(card.goalPerf, card.goalCap);
    const { value: urgency, reason: urgencyReason } = deriveUrgency(task.dueDate, now);
    const value = goal + urgency + card.impact + card.delegable;

    return {
        goal, urgency, impact: card.impact, delegable: card.delegable,
        value,
        benchmarkPoints: points,
        multiplier: card.multiplier,
        complexity,
        p: value / complexity,
        urgencyReason,
        parts: [
            { name: '大目标', value: goal,          note: `max(绩效 ${card.goalPerf}, 能力 ${card.goalCap})` },
            { name: '紧急',   value: urgency,       note: URGENCY_NOTE[urgencyReason] },
            { name: '影响',   value: card.impact,   note: IMPACT_NOTE[card.impact] },
            { name: '可委托', value: card.delegable,note: DELEGABLE_NOTE[card.delegable] },
        ],
    };
}

const URGENCY_NOTE: Record<UrgencyReason, string> = {
    'within-48h': '48h 内', 'this-week': '本周内', 'no-deadline': '无硬截止',
};
const IMPACT_NOTE = ['', '轻微', '一般', '重大'] as const;
const DELEGABLE_NOTE = ['', '可完全委托', '可部分', '必须我'] as const;
```

### 4.3 `packages/core/src/commitment-partition.ts`（新建）

```ts
import type { Task } from './types';
import type { CommitmentBenchmark, CommitmentCard } from './commitment-types';
import { computeCommitmentScore } from './commitment-score';

export interface CommitmentPartition {
    pinned: Task[];
    rated: Task[];
    unrated: Task[];
}

/**
 * 三分区。硬约束走互斥过滤（不重复出现），未评保持传入顺序。
 * 参照 focus-sections.ts:214,222,229 的成员互斥模式。
 *
 * @param defaultOrdered 已按 default 排序的任务 —— 未评任务直接沿用此顺序
 */
export function partitionByCommitment(
    defaultOrdered: readonly Task[],
    cards: Record<string, CommitmentCard>,
    benchmarks: Record<string, CommitmentBenchmark>,
    now: Date,
): CommitmentPartition {
    const pinned: Task[] = [];
    const scored: Array<{ task: Task; p: number }> = [];
    const unrated: Task[] = [];

    for (const task of defaultOrdered) {
        const card = cards[task.id];
        if (!card) { unrated.push(task); continue; }          // ← 保持传入顺序
        if (card.hardConstraints.length > 0) { pinned.push(task); continue; }

        const score = computeCommitmentScore(task, card, benchmarks[card.benchmarkId], now);
        if (!score) { unrated.push(task); continue; }          // ← 基准缺失/除零 → 视为未评
        scored.push({ task, p: score.p });
    }

    // 稳定排序：P 降序，并列时保持传入顺序
    scored.sort((a, b) => b.p - a.p);

    return { pinned, rated: scored.map((s) => s.task), unrated };
}
```

> ⚠️ **不要用 `sortTasksBySavedPreference` 的 `byId()` 兜底路径**（`task-utils.ts:1283`）—— 那会按 id 重排未评任务，违背"保持原位"。

## 13. 回滚方案

| 阶段 | 回滚方式 | 影响 |
|---|---|---|
| 步骤 0 | 单独 revert 该 commit | 无 —— 独立改动 |
| 步骤 1–2 | 删新增文件 + revert 2 个文件 | 表可保留（无引用） |
| 步骤 3 | revert 6 个文件的加值 | 排序项消失 |
| 步骤 4 | revert `focus-sections.ts` 的守卫分支 | 分区恢复 |
| 步骤 5–7 | 删新增组件 + revert 注册行 | 面板消失 |

**数据安全**：承诺卡在**独立表**里，回滚代码**不影响 `tasks` 表**。最坏情况是承诺卡数据暂时读不到，重新部署后即可恢复。
