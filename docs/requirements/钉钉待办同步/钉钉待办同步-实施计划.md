**代码骨架**

```ts
// packages/core/src/dingtalk-import.ts
import { applyImport, type ImportExecutionResult, type ImportSource, type ImportTaskSource } from './import-apply';
import { isTaskFinished } from './task-status';
import { nextRevision } from './sync-revision';
import { generateUUID as uuidv4, generateDeterministicUUID } from './uuid';
import type { AppData, Task, TaskPriority } from './types';

export const DINGTALK_IMPORT_ID_NAMESPACE = 'mindwtr:dingtalk-import:v1';
export const DINGTALK_IMPORT_SUFFIX = ' (DingTalk)';
const DINGTALK_FALLBACK = 'DingTalk';
const IMAGE_PREFIX = /^\[图片\]\s*/u;
const DONE_STAGE = 2;

export type DingTalkTodo = {
    taskId: string;
    subject: string;
    done: boolean;
    doneTime?: string | null;
    dueTime?: string | null;
    detailUrl?: string | null;
    priority?: number;
    createdAt?: string;
};

export type ParsedDingTalkImportData = {
    tasks: ImportTaskSource[];
    remoteById: Map<string, DingTalkTodo>;
    warnings: string[];
};

/** 剥离 [图片] 前缀 → 取首个非空行；为空则回退保留 [图片] 原文。 */
export const resolveDingTalkTitle = (subject: string): string => {
    const withoutImage = subject.replace(IMAGE_PREFIX, '');
    const firstLine = withoutImage.split('\n').map((l) => l.trim()).find((l) => l.length > 0);
    if (firstLine) return firstLine;
    return subject.trim() || '[图片]';
};

const toIso = (value?: string | number | null): string | undefined => {
    if (value === null || value === undefined || value === 0) return undefined;  // null 与 0 都是"无"
    const date = typeof value === 'number' ? new Date(value) : new Date(value);
    return Number.isFinite(date.getTime()) ? date.toISOString() : undefined;
};

const toPriority = (value?: number): TaskPriority => {
    // 10/20/30/40 → Mindwtr 优先级；缺失用默认
    if (value === 40) return 'urgent';
    if (value === 30) return 'high';
    if (value === 10) return 'low';
    return 'medium';
};

export const parseDingTalkTodos = (todos: readonly DingTalkTodo[]): ParsedDingTalkImportData => {
    const remoteById = new Map<string, DingTalkTodo>();
    for (const todo of todos) {
        // executor+creator 会重复返回同一条 → 按 taskId 去重，后到的覆盖先到的
        if (todo?.taskId) remoteById.set(todo.taskId, todo);
    }

    const tasks: ImportTaskSource[] = [];
    let order = 0;
    for (const todo of remoteById.values()) {
        const descriptionParts = [todo.subject.trim()];
        if (todo.detailUrl) descriptionParts.push(`钉钉详情：${todo.detailUrl}`);
        tasks.push({
            sourceKey: todo.taskId,
            title: resolveDingTalkTitle(todo.subject),
            description: descriptionParts.filter(Boolean).join('\n\n'),
            status: todo.done ? 'done' : 'inbox',
            completedAt: todo.done ? (toIso(todo.doneTime) ?? undefined) : undefined,
            dueDate: toIso(todo.dueTime),
            createdAt: toIso(todo.createdAt),
            priority: toPriority(todo.priority),
            order: order++,
            // 刻意不设 projectSourceKey / areaSourceKey → standalone（Inbox）
        });
    }
    return { tasks, remoteById, warnings: [] };
};

export const applyDingTalkImport = (
    currentData: AppData,
    parsed: ParsedDingTalkImportData,
    options: { now?: Date | string } = {},
): ImportExecutionResult & { completedExistingCount: number } => {
    const source: ImportSource = { areas: [], projects: [], tasks: parsed.tasks, warnings: parsed.warnings };

    const applied = applyImport(currentData, source, {
        fallbacks: { area: DINGTALK_FALLBACK, project: DINGTALK_FALLBACK },
        // 必传：默认是 uuidv4()，漏传则每次同步都重复建任务
        idFor: (kind, sourceKey) =>
            generateDeterministicUUID(`${DINGTALK_IMPORT_ID_NAMESPACE}:${kind}:${sourceKey}`),
        suffix: DINGTALK_IMPORT_SUFFIX,
        now: options.now,
    });

    // ---- reconcile：只回写完成态，白名单字段 ----
    const deviceId = applied.data.settings.deviceId;   // applyImport 已 ensureDeviceId
    const nowIso = (() => {
        const d = options.now instanceof Date ? options.now
            : typeof options.now === 'string' ? new Date(options.now) : new Date();
        return Number.isFinite(d.getTime()) ? d.toISOString() : new Date().toISOString();
    })();

    let completedExistingCount = 0;
    const tasks: Task[] = applied.data.tasks.map((task) => {
        const todo = parsed.remoteById.get(task.id);
        if (!todo || !todo.done) return task;
        // 跳过：墓碑 / 已取消 / 已完结（防 rev 抖动）
        if (task.deletedAt || task.purgedAt) return task;
        if (task.cancelledAt) return task;
        if (isTaskFinished(task)) return task;

        completedExistingCount += 1;
        return {
            ...task,                                    // ← 其余字段原样保留（title/dueDate 绝不动）
            status: 'done',
            completedAt: toIso(todo.doneTime) ?? nowIso, // 显式赋值，否则被回填成同步时刻
            updatedAt: nowIso,
            rev: nextRevision(task.rev),
            revBy: deviceId,
        };
    });

    return {
        ...applied,
        data: { ...applied.data, tasks },
        completedExistingCount,
    };
};
```

**关键点**：`parseDingTalkTodos` 里的 `remoteById` 用 `task.id` 查——因为 `task.id` 就是 `idFor('task', sourceKey)` 的产物，无需再算一次。

**验收标准**
- `bun --cwd packages/core test dingtalk-import` 全绿
- 覆盖设计文档 §8-A 的全部场景，特别是：
  - 连跑两次 → 第二次 `importedTaskCount === 0`
  - reconcile 后 `title` / `dueDate` **未被改动**（防回归断言）
  - `rev` 恰好 +1，`revBy === settings.deviceId`
  - 已 done / 已取消 / 墓碑 三类均不被翻转

## 4. 步骤 2：MCP 传输层

**文件**：新增 `apps/desktop/src/lib/dingtalk-mcp.ts` + `dingtalk-mcp.test.ts`

**代码骨架**（最小 streamable-HTTP JSON-RPC）

```ts
const PROTOCOL_VERSION = '2024-11-05';
const TOOL_GET_USER_TODOS = 'get_user_todos';
const PAGE_SIZE = 50;

type JsonRpcResponse = { id?: number; result?: unknown; error?: { code: number; message: string } };

const parseSse = (text: string): JsonRpcResponse[] =>
    text.split('\n\n')
        .map((frame) => frame.split('\n')
            .filter((l) => l.startsWith('data:'))
            .map((l) => l.slice(5).trim())
            .join(''))
        .filter(Boolean)
        .flatMap((payload) => { try { return [JSON.parse(payload) as JsonRpcResponse]; } catch { return []; } });

const readJsonRpc = async (response: Response): Promise<JsonRpcResponse[]> => {
    const contentType = response.headers.get('content-type') ?? '';
    const text = await response.text();
    return contentType.includes('text/event-stream') ? parseSse(text) : [JSON.parse(text) as JsonRpcResponse];
};

/** 拉取全部待办：executor+creator × 未完成+已完成，各自分页。 */
export const fetchDingTalkTodos = async (opts: {
    url: string; fetchImpl: typeof fetch; signal?: AbortSignal;
}): Promise<DingTalkTodo[]> => {
    const { url, fetchImpl, signal } = opts;
    let sessionId: string | null = null;
    let nextId = 1;

    const call = async (method: string, params: unknown, isNotification = false) => {
        const response = await fetchImpl(url, {
            method: 'POST',
            headers: {
                'content-type': 'application/json',
                accept: 'application/json, text/event-stream',
                ...(sessionId ? { 'mcp-session-id': sessionId } : {}),
            },
            body: JSON.stringify({ jsonrpc: '2.0', id: isNotification ? undefined : nextId++, method, params }),
            signal,
        });
        if (response.status === 401 || response.status === 403) throw new DingTalkAuthError();
        if (!response.ok) throw new Error(`MCP ${method} failed: HTTP ${response.status}`);
        sessionId = response.headers.get('mcp-session-id') ?? sessionId;
        const messages = await readJsonRpc(response);
        const message = messages.find((m) => m.error) ?? messages[0];
        if (message?.error) throw new Error(`MCP ${method}: ${message.error.message}`);
        return message?.result;
    };

    try {
        await call('initialize', {
            protocolVersion: PROTOCOL_VERSION,
            capabilities: {},
            clientInfo: { name: 'mindwtr', version: '1.0.0' },
        });
        await call('notifications/initialized', {}, true);

        const todos: DingTalkTodo[] = [];
        for (const roleType of ['executor', 'creator'] as const) {
            for (const todoStatus of ['false', 'true'] as const) {
                let page = 1;
                for (;;) {
                    const result = await call('tools/call', {
                        name: TOOL_GET_USER_TODOS,
                        arguments: { roleTypes: [roleType], todoStatus, pageNum: String(page), pageSize: String(PAGE_SIZE) },
                    }) as { content?: Array<{ text?: string }> } | undefined;
                    const payload = JSON.parse(result?.content?.[0]?.text ?? '{}') as {
                        result?: { todoCards?: RawCard[]; hasMore?: boolean };
                    };
                    const cards = payload.result?.todoCards ?? [];
                    todos.push(...cards.map(mapCard));
                    if (!payload.result?.hasMore) break;
                    page += 1;
                }
            }
        }
        return todos;
    } finally {
        // 会话必须关闭；streamable HTTP 可能已开 SSE
        sessionId = null;
    }
};
```

`mapCard` 负责 `finalStatusStage === 2 → done`、`dueTime` 的 `null`/`0` 归一，以及从 `detailUrl` 缺失时的兜底。

**验收标准**：本地 HTTP server 模拟（先例 `apps/mcp-server/src/http-server.test.ts:287`）→ 分页循环、SSE 分帧、超时、401 分类、会话关闭。

## 6. 步骤 4：同步编排

**文件**：新增 `apps/desktop/src/lib/dingtalk-sync.ts`；修改 `apps/desktop/src/lib/data-transfer.ts`（`export const desktopBoundaries`）

**代码骨架**

```ts
export const syncDingTalkTodos = async (opts: { now?: Date; signal?: AbortSignal } = {}): Promise<DingTalkSyncResult> => {
    const { url } = await readDingTalkMcpUrl();           // keyring
    if (!url) throw new DingTalkConfigError();

    const fetchImpl = (await getTauriHttpFetch()) ?? fetch;
    const todos = await fetchDingTalkTodos({ url, fetchImpl, signal: opts.signal });
    const parsed = parseDingTalkTodos(todos);

    const transaction = await runDataTransferTransaction({
        ...desktopBoundaries,
        operation: 'syncDingTalk',
        apply: (currentData) => {
            const applied = applyDingTalkImport(currentData, parsed, { now: opts.now });
            return { data: applied.data, result: applied };
        },
    });

    return {
        importedTaskCount: transaction.result.importedTaskCount,
        completedExistingCount: transaction.result.completedExistingCount,
        skippedExistingTaskCount: transaction.result.skippedExistingTaskCount,
        warnings: transaction.result.warnings,
    };
};
```

### 16.3 观察到的两个非 bug 现象

**① `next: 192` 是应用自身行为，不是导入错误。**
导入时全部设为 `inbox`。Mindwtr 的加载期迁移 `promote-scheduled-tasks`（`store-load-migrations.ts:317`）会把**带过期截止日期**的 inbox 任务提升为 `next`。实测分布完全吻合：

| 状态 | 有过期截止日期 | 无截止日期 | 总数 |
|---|---|---|---|
| `next` | 192 | 0 | 192 |
| `inbox` | 0 | 82 | 82 |

**② 完成时间是「同步时刻」，非钉钉勾选时刻。**
`get_user_todos` 不返回完成时间，只有 `get_todo_detail` 有 `finishTime`（逐条调用代价高）。实测差值约 53 秒。**需求方 2026-09-25 确认接受此偏差。**

### 16.4 实施中发现并修正的三个问题

| # | 问题 | 发现方式 |
|---|---|---|
| 1 | `pageSize` 上限 20，超过则**静默返回空**（`isError:false`） | 打真实网关 |
| 2 | 鉴权失败是 **HTTP 200 + JSON-RPC error body**，非 401/403 | 打真实网关 |
| 3 | 网关**间歇性返回空 result**，需重试 | 打真实网关 |
| 4 | `fully_populated_config()` 未同步新字段 → **Rust 测试编译失败**（E0063） | 读 Rust 守卫测试 |
